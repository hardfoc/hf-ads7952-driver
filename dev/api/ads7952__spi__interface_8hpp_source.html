73template <typename Port>
74inline bool Transfer(Port& port, const uint8_t* tx, uint8_t* rx, std::size_t len) {
75 using Result = TransferResult<Port>;
76 static_assert(std::is_void_v<Result> || kTransferReportsStatus<Port>,
77 "ads7952: the port's transfer(tx, rx, len) must return bool (true = frame "
78 "clocked, false = bus error), or void for a port that never reports errors");
79 if constexpr (std::is_void_v<Result>) {
80 port.transfer(tx, rx, len);
81 return true;
82 } else {
83 return static_cast<bool>(port.transfer(tx, rx, len));
